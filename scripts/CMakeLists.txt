# cmake -S scripts -B build/host-tools && cmake --build build/host-tools
# -> snx-extract + snx-recomp host binaries; no runtime deps.
cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ScarletNXScripts LANGUAGES C)
endif()
get_filename_component(snx_source_root "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)

add_library(snx-script-portable STATIC portable.c)
target_include_directories(snx-script-portable PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(WIN32)
    target_link_libraries(snx-script-portable PUBLIC shell32)
endif()
target_compile_definitions(snx-script-portable PRIVATE SNX_SOURCE_ROOT="${snx_source_root}")
add_executable(snx-extract extract.c)
add_executable(snx-recomp recomp.c)
target_link_libraries(snx-extract PRIVATE snx-script-portable)
target_link_libraries(snx-recomp PRIVATE snx-script-portable)

function(snx_script_c90 target)
    set_target_properties(${target} PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic-errors)
    elseif(MSVC)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        target_compile_options(${target} PRIVATE /W3 /TC)
    endif()
endfunction()
foreach(target snx-script-portable snx-extract snx-recomp)
    snx_script_c90(${target})
endforeach()
foreach(target snx-extract snx-recomp)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
        set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}")
    endforeach()
endforeach()
install(TARGETS snx-extract snx-recomp RUNTIME DESTINATION bin)
